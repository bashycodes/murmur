# Allowlist for tools/checks/leak-scan.py: `<path-glob> <python-regex>` per
# line. The regex is tested (re.search) against the matched text only, never
# the whole line; the glob (fnmatch) against the file path, `*` for anywhere
# including commit messages. Keep entries narrow: an explicit marker token, or
# an exact placeholder.

# Fake keys and tokens carry a marker token as a delimited segment after the
# prefix: sk-or-v1-LEAKCHECK-..., sk-or-v1-FAKE..., hf_FAKE..., a JWT signature
# .LEAKCHECK-...
* [-_.](?:LEAKCHECK|LEAK|FAKE|PROBE|SUPERSECRET)
# A fake PEM header names the marker as its key type.
* ^-----BEGIN (?:FAKE|LEAKCHECK) PRIVATE KEY-----$
# Counting fakes in older suites: a segment that opens with 0123456789 or
# abcdefgh (sk-or-v1-0123456789abcdef, sk-abcdefghijklmnop1234).
* [-_](?:0123456789|abcdefgh)
# One older fixture with its marker mid-segment, exactly (the [a] keeps this
# line from matching itself).
crates/murmur-server/tests/debug_capture.rs ^sk-or-v1-0000fake[a]pikey0000$
# AWS's own documentation key id.
* ^AKIAIOSFODNN7EXAMPLE$

# Tailnet addresses used as documentation examples, exactly.
* ^100\.64\.0\.(?:0|2|5|10|11|20|98|99)$
* ^100\.101\.102\.103$
* ^100\.100\.1\.2$
* ^100\.100\.100\.100$
* ^100\.127\.255\.255$
# Assignable range-boundary addresses, only in the classifier tests.
crates/murmur-server/src/net.rs ^100\.(?:64\.0\.1|127\.255\.254)$
apps/murmur-windows/src/dev_settings.rs ^100\.(?:64\.0\.1|127\.255\.254)$

# Placeholder accounts in docs and CI: /home/user, /home/runner (GitHub
# Actions), C:\Users\u, C:\Users\you, C:\Users\Public, CI runners.
* ^/home/(?:user|runner|you|me|USER|username|example|ubuntu|builder)$
* (?i:[\\/]users[\\/]{1,2}(?:u|me|you|user|username|public|default|example|runneradmin|circleci))$
